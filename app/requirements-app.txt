# Dependencies for the CADSmith web application.
#
# Installs the research pipeline's own requirements first, then what the app
# adds on top.  The pipeline's requirements.txt is left untouched.

-r ../requirements.txt

# Imported by autofab/metrics.py (KDTree) but missing from requirements.txt,
# so a fresh install from that file alone cannot compute CD/F1/IoU.
scipy

# Three-view rendering for the vision Judge.  autofab/render.py imports vtk
# directly; requirements.txt does not list it.
#
# On headless Linux the wheel imports fine and then fails at render time with
# no GL backend.  Install a software rasteriser as well:
#   Debian/Ubuntu:  sudo apt-get install libosmesa6
# macOS and Windows need nothing extra.  The app's health endpoint reports
# whether offscreen rendering actually works.
vtk

# Web server
fastapi
uvicorn[standard]

# Talks to OpenAI-compatible backends (OpenAI, Ollama, llama.cpp, vLLM,
# LM Studio, and the hosted gateways that speak the same API). The Anthropic
# path uses its own SDK, from requirements.txt.
httpx

# Makes Python trust the OS certificate store rather than only certifi.
# Required on corporate networks that inspect TLS, where every HTTPS
# connection is re-signed by a company CA the OS trusts and certifi does not.
truststore

# ---------------------------------------------------------------------------
# Standard part catalogue (optional)
# ---------------------------------------------------------------------------
# Both are Apache-2.0 and pinned to a commit rather than a tag: neither is
# published on PyPI, so an unpinned git URL means a demo can break when
# upstream moves. The app degrades gracefully if they are absent - the
# catalogue simply covers fewer families - so removing these two lines is a
# supported configuration, and app/tests/test_catalog_library.py checks it.
#
# cq_gears: involute spur, helical, herringbone, ring, rack and bevel gears.
# Verified: tip diameters match module x teeth + 2 modules exactly.
cq_gears @ git+https://github.com/meadiode/cq_gears.git@e73874cf17a25447a99b1e7c22a4d5af38560e9c

# cq_warehouse: 12 screw head types and 7 nut types across M1.6-M64, plus
# sprockets, roller chain and ISO threads.
#
# Two caveats, both found by testing rather than assumed. Its washers are
# broken in this revision - every standard, every size, a non-closed shell
# with roughly twice the correct volume - so washers come from
# app/catalog/parts.py instead and the router's verification step refuses to
# serve one. And it imports importlib.resources.open_text, removed in Python
# 3.13, so this pin needs Python 3.11 or 3.12.
cq_warehouse @ git+https://github.com/gumyr/cq_warehouse.git@daa46507ecc429c0e2dce11d9d5ffd09b12a42af
